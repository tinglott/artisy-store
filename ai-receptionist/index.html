receptionist.py 
import subprocess, time, json
from datetime import datetime

# ── Load business knowledge (RAG) ──────────────────────
with open("knowledge.json", "r") as f:
    knowledge = json.load(f)

def stt_audio():
    """Transcribe call audio via Whisper.cpp"""
    subprocess.run([
        "./whisper.cpp/main", "-m", "models/ggml-base.en.bin",
        "-f", "call.wav", "--output-json"
    ])
    with open("call.wav.json") as f:
        return json.load(f)["text"]

def llm_generate(prompt):
    """Generate response with local Phi-3"""
    result = subprocess.run([
        "./llama.cpp/llama-cli", "-m", "Phi-3-mini-4k-instruct-q4_0.gguf",
        "--prompt", prompt, "-n", "512", "--temp", "0.7"
    ], capture_output=True, text=True)
    return result.stdout

def tts_speak(text):
    """Convert response to speech via Piper TTS"""
    subprocess.run([
        "piper", "--model", "en-us-amy-medium.onnx",
        "--output_file", "response.wav"
    ], input=text, text=True)
    subprocess.run(["termux-media-player", "play", "response.wav"])

# ── Main loop (triggered by PhoneClaw / Tasker) ─────────
while True:
    text = stt_audio()
    context = f"Business: {knowledge}\nCaller said: {text}\nRespond helpfully:"
    if "book" in text.lower() or "appointment" in text.lower():
        tts_speak("Sure! Let me check the calendar for you.")
        # PhoneClaw Accessibility opens Calendar + books slot
    response = llm_generate(context)
    tts_speak(response)
    time.sleep(1)

Production Tips

⚡
Use a Pixel or Samsung for NPU Speed
Newer Pixel/Samsung phones with NPU run quantized models 2–4× faster. A Pixel 7a or Galaxy A54 is the sweet spot — affordable and fast.
🔋
Keep the Phone Plugged In
Use termux-wake-lock + nohup to keep the pipeline running 24/7. Plug into a USB charger so the battery never drains during active calls.
🗄️
Hybrid Storage Option
For full RAG + history beyond phone storage, self-host PocketBase (single binary) or lightweight SQLite + sync to Google Drive via Termux.
🔄
Duplex Conversation
For real-time back-and-forth, combine with the talk-llama example in whisper.cpp. Enables natural duplex conversation without push-to-talk.
📊
Monitor via Local Dashboard
Run the Node.js dashboard on your phone (Termux) and access it from your laptop on the same WiFi. Full call logs, appointment history, and status.
🧠
Pre-Load Your Knowledge Base
Scrape your website once → embed locally with BAAI/bge-small-en-v1.5 → store in Chroma. The AI will answer any FAQ about your business instantly.
